let parts = msg.payload.split(",").map(Number);

// แปลงเลข 32-bit → array บิต
function toBits32(num) {
    let bits = [];
    for (let i = 0; i < 32; i++) {
        bits.push((num >> i) & 1);
    }
    return bits;
}

let xBits = toBits32(parts[0]).concat(toBits32(parts[1])); // X0–X77
let yBits = toBits32(parts[2]).concat(toBits32(parts[3])); // Y0–Y77

msg.payload = {
    X: xBits,   // 64 จุด
    Y: yBits    // 64 จุด
};

return msg;
